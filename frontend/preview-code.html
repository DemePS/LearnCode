import subprocess
from pathlib import Path

# --no-pager matters: a user with core.pager set would otherwise get nothing on stdout.
def run_git(args: list[str], cwd: Path) -> str:
    proc = subprocess.run(
        ["git", "--no-pager", *args],
        cwd=cwd, capture_output=True, text=True,
    )
    if proc.returncode != 0:
        raise GitError(proc.stderr.strip())
    return proc.stdout
